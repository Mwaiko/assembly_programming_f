section .data
    prompt     db "Enter your name: "
    prompt_len equ $ - prompt

    output     db "Hello, "
    output_len equ $ - output

section .bss
    input_buf resb 64

section .text
    global _start

_start:
    ; 1. Display prompt
    mov rax, 1.0
    mov rdi, 1
    mov rsi, prompt
    mov rdx, prompt_len
    syscall

    ; 2. Read user's name
    mov rax, 0
    xor rdi, rdi 
    mov rsi, input_buf
    mov rdx, 63
    syscall

    ; RAX now contains the number of bytes read.
    ; Save it because the next syscall will overwrite RAX.
    mov r12, rax

    ; 3. Display greeting prefix
    mov rax, 1
    mov rdi, 1
    mov rsi, output
    mov rdx, output_len
    syscall

    ; 4. Display the actual input
    mov rax, 1
    mov rdi, 1
    mov rsi, input_buf
    mov rdx, r12             ; Reuse bytes returned by sys_read
    syscall

    ; 5. Exit
    mov rax, 60
    xor rdi, rdi
    syscall



section .data
    prompt     db "Enter your name: "
    prompt_len equ $ - prompt

    output     db "Hello, "
    output_len equ $ - output

section .bss
    input_buf resb 64

section .text
    global _start

_start:
    ; 1. Display prompt (sys_write)
    mov edx, prompt_len
    mov ecx, prompt
    mov ebx, 1
    mov eax, 4
    int 0x80

    ; 2. Read user's name (sys_read)
    mov edx, 63
    mov ecx, input_buf
    mov ebx, 0
    mov eax, 3
    int 0x80

    ; EAX contains the number of bytes read.
    ; Save the value before another syscall overwrites EAX.
    mov esi, eax

    ; 3. Display greeting prefix (sys_write)
    mov edx, output_len
    mov ecx, output
    mov ebx, 1
    mov eax, 4
    int 0x80

    ; 4. Display the user's actual input
    mov edx, esi             ; Reuse bytes returned by sys_read
    mov ecx, input_buf
    mov ebx, 1
    mov eax, 4
    int 0x80

    ; 5. Exit program
    xor ebx, ebx
    mov eax, 1
    int 0x80

section .data
    filename      db "test.txt", 0

    text_to_write db "Hello from NASM 64-bit file IO!", 0xa
    text_len      equ $ - text_to_write

    err_msg       db "File operation failed!", 0xa
    err_len       equ $ - err_msg

section .bss
    read_buf      resb 64

section .text
    global _start

_start:

    ; 1. OPEN / CREATE FILE FOR WRITING
    mov rax, 2                 ; sys_open
    mov rdi, filename
    mov rsi, 577               ; O_WRONLY | O_CREAT | O_TRUNC
    mov rdx, 420               ; 0644 permissions
    syscall

    test rax, rax
    js .error

    ; Save file descriptor
    mov r12, rax

    ; 2. WRITE DATA TO FILE
    mov rax, 1                 ; sys_write
    mov rdi, r12               ; Reuse file descriptor
    mov rsi, text_to_write
    mov rdx, text_len
    syscall

    test rax, rax
    js .error

    ; 3. CLOSE FILE
    mov rax, 3                 ; sys_close
    mov rdi, r12
    syscall

    test rax, rax
    js .error

    ; 4. OPEN FILE FOR READING
    mov rax, 2                 ; sys_open
    mov rdi, filename
    xor rsi, rsi               ; O_RDONLY
    xor rdx, rdx
    syscall

    test rax, rax
    js .error

    ; Save the new file descriptor
    mov r12, rax

    ; 5. READ FILE CONTENT
    mov rax, 0                 ; sys_read
    mov rdi, r12
    mov rsi, read_buf
    mov rdx, 64                ; Maximum bytes to read
    syscall

    test rax, rax
    js .error

    ; Save number of bytes actually read
    mov r13, rax

    ; 6. DISPLAY CONTENT ON SCREEN
    mov rax, 1                 ; sys_write
    mov rdi, 1                 ; stdout
    mov rsi, read_buf
    mov rdx, r13               ; Reuse bytes returned by sys_read
    syscall

    ; 7. CLOSE FILE
    mov rax, 3                 ; sys_close
    mov rdi, r12
    syscall

    ; 8. EXIT
    mov rax, 60
    xor rdi, rdi
    syscall


.error:
    ; Display error message
    mov rax, 1
    mov rdi, 1
    mov rsi, err_msg
    mov rdx, err_len
    syscall

    ; Exit with error status
    mov rax, 60
    mov rdi, 1
    syscall

section .data
    filename       db "log.txt", 0

    new_line_text  db "Appended log entry.", 0xa
    text_len       equ $ - new_line_text

    header_msg     db "--- Reading File Contents ---", 0xa
    header_len     equ $ - header_msg

    err_msg        db "File operation failed!", 0xa
    err_len        equ $ - err_msg

section .bss
    read_buf       resb 256

section .text
    global _start

_start:

    ; 1. OPEN FILE FOR APPENDING
    mov rax, 2                 ; sys_open
    mov rdi, filename
    mov rsi, 1089              ; O_WRONLY | O_CREAT | O_APPEND
    mov rdx, 420               ; 0644 permissions
    syscall

    test rax, rax
    js .error

    ; Save returned file descriptor
    mov r12, rax

    ; 2. APPEND DATA TO FILE
    mov rax, 1                 ; sys_write
    mov rdi, r12               ; Reuse file descriptor
    mov rsi, new_line_text
    mov rdx, text_len
    syscall

    test rax, rax
    js .error

    ; 3. CLOSE FILE
    mov rax, 3                 ; sys_close
    mov rdi, r12
    syscall

    test rax, rax
    js .error

    ; 4. DISPLAY HEADER
    mov rax, 1                
    mov rdi, 1              
    mov rsi, header_msg
    mov rdx, header_len
    syscall

    ; 5. OPEN FILE FOR READING
    mov rax, 2        
    mov rdi, filename
    xor rsi, rsi               ; O_RDONLY
    xor rdx, rdx
    syscall

    test rax, rax
    js .error

    ; Save new file descriptor
    mov r12, rax

    ; 6. READ FILE CONTENTS
    mov rax, 0      
    mov rdi, r12
    mov rsi, read_buf
    mov rdx, 256       
    syscall

    test rax, rax
    js .error

    ; Save number of bytes actually read
    mov r13, rax

    ; 7. DISPLAY FILE CONTENTS
    mov rax, 1               
    mov rdi, 1          
    mov rsi, read_buf
    mov rdx, r13               ; Reuse bytes returned by sys_read
    syscall

    ; 8. CLOSE FILE
    mov rax, 3       
    mov rdi, r12
    syscall

    test rax, rax
    js .error

    ; 9. EXIT SUCCESSFULLY
    mov rax, 60
    xor rdi, rdi
    syscall


.error:
    ; Display error message
    mov rax, 1
    mov rdi, 1
    mov rsi, err_msg
    mov rdx, err_len
    syscall

    ; Exit with error status
    mov rax, 60
    mov rdi, 1
    syscall
